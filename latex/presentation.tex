\begin{frame}
    \frametitle{Diferença das Gaussianas Baseada em Fluxo}
    \framesubtitle{Tensor de estrutura}

    \begin{equation*}
        S =
        \begin{bmatrix} G_x * G_x & G_x * G_y \\ G_x * G_x & G_y * G_y \end{bmatrix} =
        \begin{bmatrix} E & F \\ F & G \end{bmatrix}
    \end{equation*}
\end{frame}

\begin{frame}
    \frametitle{Diferença das Gaussianas Baseada em Fluxo}
    \framesubtitle{Vetor tangente}

    \begin{equation*}
        S =
        \begin{bmatrix} G_x * G_x & G_x * G_y \\ G_x * G_y & G_y * G_y \end{bmatrix} =
        \begin{bmatrix} E & F \\ F & G \end{bmatrix}
    \end{equation*}

    \begin{equation*}
        \hat{t} = \begin{pmatrix} \lambda_1 - E \\ -F \end{pmatrix}
    \end{equation*}
\end{frame}
